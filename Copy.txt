pipeline {
    agent any

    stages {

        /* -------------------------
         * REPO 1 CHECKOUT
         * ------------------------- */
        stage('Checkout Repo 1') {
            steps {
                dir('repo-1') {
                    checkout([
                        $class: 'GitSCM',
                        branches: [[name: '*/main']],    // <-- BRANCH 1
                        userRemoteConfigs: [[
                            url: 'https://github.com/org/repo1.git',  // <-- REPO 1
                            credentialsId: 'git-cred'
                        ]]
                    ])
                }
            }
        }

        /* -------------------------
         * REPO 2 CHECKOUT
         * ------------------------- */
        stage('Checkout Repo 2') {
            steps {
                dir('repo-2') {
                    checkout([
                        $class: 'GitSCM',
                        branches: [[name: '*/develop']],   // <-- BRANCH 2
                        userRemoteConfigs: [[
                            url: 'https://github.com/org/repo2.git',  // <-- REPO 2
                            credentialsId: 'git-cred'
                        ]]
                    ])
                }
            }
        }

        /* -------------------------
         * REPO 3 CHECKOUT
         * ------------------------- */
        stage('Checkout Repo 3') {
            steps {
                dir('repo-3') {
                    checkout([
                        $class: 'GitSCM',
                        branches: [[name: '*/feature/api']],  // <-- BRANCH 3
                        userRemoteConfigs: [[
                            url: 'https://github.com/org/repo3.git', // <-- REPO 3
                            credentialsId: 'git-cred'
                        ]]
                    ])
                }
            }
        }

        /* -------------------------
         * EXECUTE SHELL STEP
         * ------------------------- */
        stage('Execute Shell') {
            steps {
                sh '''
                echo "=== CURRENT WORKSPACE CONTENT ==="
                ls -la

                echo "=== Inside repo-1 ==="
                cd repo-1
                ls -la

                echo "=== Inside repo-2 ==="
                cd ../repo-2
                ls -la

                echo "=== Inside repo-3 ==="
                cd ../repo-3
                ls -la

                echo "Shell executed successfully!"
                '''
            }
        }
    }
}
