// public/runtime-shim.js
(function () {
  const CFG = window.RUNTIME_CONFIG || {};
  const TOKENS = Object.keys(CFG).map(k => [`__${k}__`, CFG[k]]);

  // Fastest replacement (no loops inside fetch/XHR)
  function rewrite(str) {
    if (typeof str !== "string") return str;
    for (let i = 0; i < TOKENS.length; i++) {
      const [token, value] = TOKENS[i];
      if (str.includes(token)) {
        str = str.replaceAll(token, value);
      }
    }
    return str;
  }

  // PATCH FETCH (super lightweight)
  if (window.fetch) {
    const nativeFetch = window.fetch;
    window.fetch = function (input, init) {
      if (typeof input === "string") input = rewrite(input);
      else if (input && input.url) input.url = rewrite(input.url);
      return nativeFetch(input, init);
    };
  }

  // PATCH XHR
  const XHROpen = XMLHttpRequest.prototype.open;
  XMLHttpRequest.prototype.open = function (m, url) {
    return XHROpen.call(this, m, rewrite(url), ...[].slice.call(arguments, 2));
  };

  // PATCH AXIOS (optional but safe)
  if (window.axios && window.axios.interceptors) {
    window.axios.interceptors.request.use(function (config) {
      if (config.url) config.url = rewrite(config.url);
      return config;
    });
  }
})();
